    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Add a default-on tqdm progress bar to every long-running kriging/estimation, simulation, I/O and modeling call exposed to Python, without changing any numeric output.

    Architecture: A single Progress counter type in ceres-core (an AtomicU64 side-channel, never touching the parallel reduction it accompanies) is threaded through the small number of shared driver functions that already implement rayon parallelism (estimation::batch::estimate_many, estimation::lva::estimate_many_local, estimation::indicator::indicator_targets, simulation::post::run, and a few standalone loops in the pyo3 bindings). A single pyo3 helper, crate::progress::with_progress, wraps each Python-facing call: when progress=True it builds a tqdm bar and drives it from a scoped monitor thread while the real computation runs with the GIL released; when progress=False it is a zero-overhead passthrough.

    Tech Stack: Rust workspace (pyo3 0.29, rayon 1.10, arrow/parquet), Python bindings via maturin, tqdm (new core Python dependency), pytest.

    use std::sync::atomic::{AtomicU64, Ordering};

/// Counts units of work done. `total` is `None` when it isn't known upfront
/// (e.g. a text-format read whose row count isn't known before scanning it).
pub struct Progress {
    done: AtomicU64,
    total: Option<u64>,
}

impl Progress {
    pub fn new(total: Option<u64>) -> Self {
        Self {
            done: AtomicU64::new(0),
            total,
        }
    }

    /// Records one unit of work done.
    pub fn inc(&self) {
        self.done.fetch_add(1, Ordering::Relaxed);
    }

    /// Records `n` units of work done at once.
    pub fn inc_by(&self, n: u64) {
        self.done.fetch_add(n, Ordering::Relaxed);
    }

    /// Units done so far, and the total when known upfront.
    pub fn snapshot(&self) -> (u64, Option<u64>) {
        (self.done.load(Ordering::Relaxed), self.total)
    }
}

    //! Shared driver for the `progress` kwarg: builds and updates a `tqdm` bar
//! from a background thread while the real computation runs with the GIL
//! released, so the parallel work is never serialized behind Python calls.

use std::sync::atomic::{AtomicBool, Ordering};
use std::thread;
use std::time::Duration;

use ceres_core::Progress;
use pyo3::prelude::*;
use pyo3::types::PyDict;

/// Runs `f` with a `Progress` counter of `total` units (`None` when unknown
/// upfront). When `show`, a `tqdm` bar tracks it; when not, `f` runs the same
/// way with no counter and no extra thread, at zero cost.
pub fn with_progress<T: Send>(
    py: Python<'_>,
    total: Option<u64>,
    show: bool,
    f: impl FnOnce(Option<&Progress>) -> T + Send,
) -> PyResult<T> {
    if !show {
        return Ok(py.detach(|| f(None)));
    }
    let kwargs = PyDict::new(py);
    if let Some(total) = total {
        kwargs.set_item("total", total)?;
    }
    let bar = py
        .import("tqdm")?
        .getattr("tqdm")?
        .call((), Some(&kwargs))?
        .unbind();
    let counter = Progress::new(total);
    let stop = AtomicBool::new(false);
    let result = py.detach(|| {
        thread::scope(|scope| {
            let monitor = scope.spawn(|| {
                let mut last = 0u64;
                while !stop.load(Ordering::Relaxed) {
                    thread::park_timeout(Duration::from_millis(50));
                    let (done, _) = counter.snapshot();
                    if done > last {
                        let sent = Python::attach(|py| {
                            bar.bind(py).call_method1("update", (done - last,))
                        });
                        if let Err(e) = sent {
                            Python::attach(|py| e.print(py));
                            return;
                        }
                        last = done;
                    }
                }
                let (done, _) = counter.snapshot();
                if done > last {
                    let _ = Python::attach(|py| {
                        bar.bind(py).call_method1("update", (done - last,))
                    });
                }
            });
            let result = f(Some(&counter));
            stop.store(true, Ordering::Relaxed);
            monitor.thread().unpark();
            let _ = monitor.join();
            result
        })
    });
    Python::attach(|py| {
        let _ = bar.bind(py).call_method0("close");
    });
    Ok(result)
}

    use ceres_core::Progress;

/// Reads a GSLIB file: title, column count, one name per line, then
/// whitespace-separated rows. Extra tokens on a row are ignored. The title is
/// kept in the schema metadata under `title`. `progress`, when given, is
/// ticked once per row; the total is not known upfront for a text format.
pub fn read_gslib(
    path: impl AsRef<Path>,
    nodata: &[Nodata],
    progress: Option<&Progress>,
) -> Result<RecordBatch> {
    let text = std::fs::read_to_string(path)?;
    let mut lines = text
        .lines()
        .enumerate()
        .map(|(i, l)| (i + 1, l.trim()))
        .filter(|(_, l)| !l.is_empty());
    let bad = |line, message: &str| Error::Format {
        line,
        message: message.into(),
    };

    let (_, title) = lines.next().ok_or_else(|| bad(1, "missing title"))?;
    let (line, count) = lines.next().ok_or_else(|| bad(2, "missing column count"))?;
    let count: usize = count
        .split_whitespace()
        .next()
        .and_then(|n| n.parse().ok())
        .ok_or_else(|| bad(line, "column count is not an integer"))?;
    let names: Vec<&str> = lines.by_ref().take(count).map(|(_, l)| l).collect();
    if names.len() < count {
        return Err(bad(line, "fewer column names than declared"));
    }

    let mut columns = vec![Vec::new(); count];
    for (line, row) in lines {
        let tokens: Vec<&str> = row.split_whitespace().collect();
        if tokens.len() < count {
            return Err(bad(
                line,
                &format!("expected {count} values, found {}", tokens.len()),
            ));
        }
        for (column, token) in columns.iter_mut().zip(tokens) {
            let value = if is_nodata(token, nodata) {
                None
            } else {
                Some(
                    token
                        .parse::<f64>()
                        .map_err(|_| bad(line, &format!("`{token}` is not a number")))?,
                )
            };
            column.push(value);
        }
        if let Some(p) = progress {
            p.inc();
        }
    }

    let rows = columns.first().map_or(0, Vec::len);
    let fields: Vec<Field> = names
        .iter()
        .map(|n| Field::new(*n, DataType::Float64, true))
        .collect();
    let metadata = HashMap::from([("title".to_string(), title.to_string())]);
    let columns: Vec<ArrayRef> = columns
        .into_iter()
        .map(|c| Arc::new(Float64Array::from(c)) as ArrayRef)
        .collect();
    Ok(RecordBatch::try_new_with_options(
        Arc::new(Schema::new_with_metadata(fields, metadata)),
        columns,
        &RecordBatchOptions::new().with_row_count(Some(rows)),
    )?)
}

/// Writes numeric columns as GSLIB; nulls are written as `nodata`. `progress`,
/// when given, is ticked once per row; the total is `table.num_rows()`.
pub fn write_gslib(
    path: impl AsRef<Path>,
    table: &RecordBatch,
    nodata: f64,
    progress: Option<&Progress>,
) -> Result<()> {
    let schema = table.schema();
    if let Some(f) = schema.fields().iter().find(|f| !f.data_type().is_numeric()) {
        return Err(Error::NotNumeric(f.name().clone()));
    }
    let columns = table
        .columns()
        .iter()
        .map(|c| cast(c, &DataType::Float64))
        .collect::<std::result::Result<Vec<_>, _>>()?;
    let columns: Vec<&Float64Array> = columns
        .iter()
        .map(|c| c.as_primitive::<Float64Type>())
        .collect();

    let mut out = BufWriter::new(File::create(path)?);
    let title = schema
        .metadata()
        .get("title")
        .map_or("ceres", String::as_str);
    writeln!(out, "{title}\n{}", columns.len())?;
    for field in schema.fields() {
        writeln!(out, "{}", field.name())?;
    }
    for row in 0..table.num_rows() {
        let values: Vec<String> = columns
            .iter()
            .map(|c| {
                if c.is_valid(row) {
                    c.value(row)
                } else {
                    nodata
                }
                .to_string()
            })
            .collect();
        writeln!(out, "{}", values.join(" "))?;
        if let Some(p) = progress {
            p.inc();
        }
    }
    Ok(out.flush()?)
}

/// given, is ticked by each chunk's row count; the total is `table.num_rows()`.
pub fn write_csv(
    path: impl AsRef<Path>,
    table: &RecordBatch,
    progress: Option<&Progress>,
) -> Result<()> {
    let mut writer = WriterBuilder::new()
        .with_header(true)
        .build(BufWriter::new(File::create(path)?));
    let rows = table.num_rows();
    let mut offset = 0;
    loop {
        let len = CHUNK.min(rows - offset);
        writer.write(&table.slice(offset, len))?;
        offset += len;
        if let Some(p) = progress {
            p.inc_by(len as u64);
        }
        if offset >= rows {
            break;
        }
    }

    fn parse_stl(bytes: &[u8], progress: Option<&Progress>) -> Result<Mesh> {
    let mut corners = Vec::new();
    let count = bytes
        .get(80..84)
        .map(|b| u32::from_le_bytes(b.try_into().expect("4 bytes")) as usize);
    if count.is_some_and(|n| bytes.len() == 84 + 50 * n) {
        for record in bytes[84..].chunks_exact(50) {
            let f = |i: usize| f32::from_le_bytes(record[i..i + 4].try_into().expect("4")) as f64;
            corners.extend((0..3).map(|v| [0, 1, 2].map(|a| f(12 + 12 * v + 4 * a))));
            if let Some(p) = progress {
                p.inc();
            }
        }
    } else {
        let bad = || Error::Mesh("not a binary or ASCII STL file".into());
        let text = std::str::from_utf8(bytes).map_err(|_| bad())?;
        if !text.trim_start().starts_with("solid") {
            return Err(bad());
        }
        let mut tokens = text.split_whitespace();
        while let Some(token) = tokens.next() {
            if token == "vertex" {
                let mut p = [0.0; 3];
                for x in &mut p {
                    *x = tokens
                        .next()
                        .and_then(|t| t.parse::<f32>().ok())
                        .ok_or_else(bad)? as f64;
                }
                corners.push(p);
            }
        }
        if corners.len() % 3 != 0 {
            return Err(bad());
        }
    }
    weld(&corners)
}

    /// Runs realizations `0..n` in parallel batches and hands them to `add` in
/// order; at most one batch (one realization per thread) is held at a time.
/// `progress`, when given, is ticked once per realization.
fn run<R: Send>(
    n: usize,
    simulate: impl Fn(usize) -> Result<R> + Sync,
    mut add: impl FnMut(R) -> Result<()>,
    progress: Option<&Progress>,
) -> Result<()> {
    if n == 0 {
        return Err(SimError::InvalidParameters(
            "need at least one realization".into(),
        ));
    }
    let batch = rayon::current_num_threads().max(1);
    for start in (0..n).step_by(batch) {
        let done: Vec<R> = (start..(start + batch).min(n))
            .into_par_iter()
            .map(&simulate)
            .collect::<Result<_>>()?;
        let count = done.len();
        done.into_iter().try_for_each(&mut add)?;
        if let Some(p) = progress {
            p.inc_by(count as u64);
        }
    }
    Ok(())
}
