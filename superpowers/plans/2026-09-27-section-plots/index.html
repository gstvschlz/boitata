    True-edge section plots Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: section() draws a block model's true cell edges (not a resampled raster) whenever the cutting plane is normal to one of the model's own local axes, on a new BlockModel.corners() primitive; slab() is unchanged, and the two compose on one ax via a shared plane tuple for full validation plates (true edges + lens trace + drillhole composites, one colour scale).

    Architecture: One new Rust method (BlockModel::corners, 8 world-space corners per row) exposed as a pyo3 property. section()'s Python internals gain a small geometry layer (model-local-axis vectors from the existing _principal rotation helper, a plane-vs-axis alignment check, a PatchCollection renderer) that sits beside the existing raster path, which stays as the fallback for a plane at a genuine angle to the model. No public signature changes anywhere.

    Tech Stack: Rust (ceres-core, nalgebra), pyo3/numpy bindings (ceres-py), Python/matplotlib (ceres.plot), pytest, proptest.

    Global Constraints¶

    • Rust: no unsafe, thiserror errors only, no panics on user input (AGENTS.md).
    • Rust/Python parity: every feature ships in both languages in the same PR (AGENTS.md, Definition of Done).
    • One theory check per new algorithm (AGENTS.md, Definition of Done): here, corner-box volume == volumes().
    • .pyi stub and NumPy docstrings updated alongside any binding change (AGENTS.md, Definition of Done).
    • One gallery example in examples/ (AGENTS.md, Definition of Done).
    • Conventional commits, no AI attribution in commit messages, branch + PR to main, never commit to main directly (AGENTS.md).
    • Never add a dependency for what a few lines do (AGENTS.md, CLAUDE.md) — this plan adds none.
    • American spelling, existing naming conventions (coords, values, data=, search, plane) (AGENTS.md).

    In crates/core/src/block_model.rs, right after volumes() (which ends at line 381, just before /// Keeps the rows where \keep` is true.../pub fn mask`):

        /// World-space corners of each row's box: 8 vertices. Vertex `c` (0..8)
    /// takes, for axis `a` (0..3), the row's minimum extent on that axis if
    /// bit `a` of `c` is 0, its maximum if 1 — so `c = 0` is the all-minimum
    /// corner, `c = 7` the all-maximum one. Rotation-aware via
    /// [`Geometry::point`].
    pub fn corners(&self) -> Vec<[[f64; 3]; 8]> {
        let whole = [0.0, 0.0, 0.0, 1.0, 1.0, 1.0];
        match &self.layout {
            Layout::SubBlocked { parent, extent, .. } => parent
                .iter()
                .zip(extent)
                .map(|(&p, e)| corners_of(&self.geometry, p, e))
                .collect(),
            _ => (0..self.len())
                .map(|row| corners_of(&self.geometry, self.parent_index(row), &whole))
                .collect(),
        }
    }

    Right after rows() (the last free function in the file, currently ending at line 723), before #[cfg(test)]:

    fn corners_of(geometry: &Geometry, index: u64, extent: &[f64; 6]) -> [[f64; 3]; 8] {
    let mut corners = [[0.0; 3]; 8];
    for (c, corner) in corners.iter_mut().enumerate() {
        let at = [0, 1, 2].map(|a| extent[a + 3 * ((c >> a) & 1)]);
        *corner = geometry.point(index, at);
    }
    corners
}

    def _block_axes(rotation):
    """Model's local x, y, z unit vectors in world space, rows of a (3, 3) matrix."""
    p = _principal(rotation)
    return np.array([-p[1], p[0], p[2]])


def _angles_for_normal(axis):
    """(azimuth, dip) for `_frame` whose plane normal is `axis` (a unit vector)."""
    dip = np.degrees(np.arccos(np.clip(axis[2], -1.0, 1.0)))
    spread = np.hypot(axis[0], axis[1])
    azimuth = 90.0 if spread < 1e-12 else np.degrees(np.arctan2(-axis[1], axis[0]))
    return azimuth, dip


def _axis_plane(model, axis, index):
    """`plane` tuple through cell `index` along `axis`, normal to that local axis."""
    k = {"x": 0, "y": 1, "z": 2}[axis]
    size, count = np.asarray(model.size, dtype=float), np.asarray(model.count, dtype=float)
    index = count[k] // 2 if index is None else float(index)
    local = size * count / 2
    local[k] = (index + 0.5) * size[k]
    axes = _block_axes(model.rotation)
    center = np.asarray(model.origin) + local @ axes
    azimuth, dip = _angles_for_normal(axes[k])
    return tuple(center), azimuth, dip


def _aligned_axis(model, plane):
    """Local axis (0, 1, 2) `plane` is normal to, or None if it isn't axis-aligned."""
    n = _frame(plane)[3]
    axes = _block_axes(model.rotation)
    matches = np.flatnonzero(np.abs(axes @ n) > 1 - 1e-9)
    return int(matches[0]) if len(matches) else None


def _blocks(ax, model, values, plane, k, scheme, kwargs):
    """True block-edge rendering: draws every row whose box straddles `plane`, as a `PatchCollection`."""
    from matplotlib.collections import PatchCollection
    from matplotlib.patches import Polygon

    center, u, v, n = _frame(plane)
    corners = model.corners
    axis = _block_axes(model.rotation)[k]
    depth = (corners - center) @ axis
    keep = (depth.min(axis=1) <= 0) & (depth.max(axis=1) >= 0)
    if not keep.any():
        raise InvalidInput("the plane misses the block model")
    others = [a for a in range(3) if a != k]
    face = [sum(bit << others[i] for i, bit in enumerate(bits)) for bits in ((0, 0), (0, 1), (1, 1), (1, 0))]
    faces = corners[keep][:, face, :] @ np.c_[u, v]
    values = np.asarray(_column(model, values), dtype=float)[keep]
    _scheme_colors(scheme, kwargs)

    """
# 86. Section validation plates

A block model earns trust one section at a time: true block edges (not a resampled raster, so a sub-blocked
model shows its real geometry), the estimate on those blocks, the lens it was built from, and the drillhole
composites it was built on, all on one plate and one colour scale. `section` draws true edges whenever the
cut is normal to one of the model's own axes; `slab` overlays the lens trace and the composites on the same
plane. Composite-in-block adherence — does the model honour the holes it was built from — is a `row_at`
lookup away from `scatter`.
"""

# %% [hidden]
import sys
from pathlib import Path

HERE = Path(__file__).parent if "__file__" in globals() else Path.cwd()
sys.path.insert(0, str(HERE.parents[1]))

# %%
import ceres as cs
import matplotlib.pyplot as plt
import numpy as np
from common import save

# %% [markdown]
# Sub-block the parent grid to one lens (topic 50), then krige a grade onto the sub-blocks from the drillhole
# composites inside it.

# %%
data = cs.datasets.stacked_sulphide_lenses()
lens = data["lens_2"]
parents = cs.BlockModel.from_extents(lens, size=(40, 40, 20), buffer=10, snap=True)
model = parents.subblock([(lens, "inside", "ore")], 4, fill="waste")

holes = cs.Drillholes(data["collars"], data["surveys"], data["assays"])
composites = holes.composite(2.0, ["ZN_PCT"])
xyz, zn = composites.coords, composites["ZN_PCT"]
inside = lens.contains(xyz) & ~np.isnan(zn)
search = cs.Search(100, max_samples=12)
domain = np.array(model["domain"])
ore = domain == "ore"
grade = np.full(len(model), np.nan)
grade[ore] = cs.InverseDistance(search).fit(xyz[inside], zn[inside]).predict(model.centroids[ore])
model = model.with_column("zn", grade)

# %% [markdown]
# `section` draws true block edges whenever the cut is normal to one of the model's own axes — here an
# explicit `plane` chosen across the lens's own strike. `slab` overlays the lens trace and the composite
# pierce points on that very same `plane`, on the same colour scale.

# %%
plane = (lens.vertices.mean(axis=0), 112.5, 90)
fig, ax = plt.subplots(figsize=(9, 6), layout="constrained")
cs.plot.section(model, "zn", plane=plane, cmap="viridis", vmin=0, vmax=8, ax=ax)
cs.plot.slab(
    composites, "ZN_PCT", plane=plane, thickness=25, meshes=lens, cmap="viridis", vmin=0, vmax=8,
    colorbar=False, ax=ax,
)
ax.set(title="Section across strike: true sub-block edges, lens trace, composite pierce points")
save(fig, "section")

# %% [markdown]
# ## Adherence
#
# Adherence is a resubstitution test, not cross-validation: does the model honour the holes it was built
# from? `row_at` finds the sub-block holding each composite; `scatter` gives the 1:1 line and the regression
# slope.

# %%
rows = model.row_at(xyz[inside])
paired = rows >= 0
fig, ax = plt.subplots(figsize=(5, 5), layout="constrained")
cs.plot.scatter(zn[inside][paired], model["zn"][rows[paired]], ax=ax)
ax.set(xlabel="composite Zn (%)", ylabel="block Zn (%)", title=f"Adherence, {paired.sum()} composites")
save(fig, "adherence")
